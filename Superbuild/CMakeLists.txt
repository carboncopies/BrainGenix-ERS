# Set CMAKE Version, Define Project
cmake_minimum_required(VERSION 3.15)

project(
    BrainGenix-ERS-Superbuild
    VERSION 0.1.0
    LANGUAGES NONE
)

include(ExternalProject)

set(ERS_SUPERBUILD_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/..")
get_filename_component(ERS_SUPERBUILD_SOURCE_DIR "${ERS_SUPERBUILD_SOURCE_DIR}" REALPATH)

set(ERS_SUPERBUILD_BINARY_DIR "${CMAKE_BINARY_DIR}/ers-build")
set(ERS_SUPERBUILD_INSTALL_DIR "${CMAKE_BINARY_DIR}/ers-install")
set(ERS_SUPERBUILD_TOOLCHAIN_FILE "${ERS_SUPERBUILD_SOURCE_DIR}/ThirdParty/vcpkg/scripts/buildsystems/vcpkg.cmake")

option(ERS_SUPERBUILD_ENABLE_PACKAGING "Enable CPack install prefix wiring for the first-pass superbuild entrypoint" OFF)

set(ERS_SUPERBUILD_CACHE_ARGS
    "-DCMAKE_TOOLCHAIN_FILE:FILEPATH=${ERS_SUPERBUILD_TOOLCHAIN_FILE}"
    "-DCMAKE_INSTALL_PREFIX:PATH=${ERS_SUPERBUILD_INSTALL_DIR}"
)

if (CMAKE_BUILD_TYPE)
    list(APPEND ERS_SUPERBUILD_CACHE_ARGS "-DCMAKE_BUILD_TYPE:STRING=${CMAKE_BUILD_TYPE}")
endif()

if (ERS_SUPERBUILD_ENABLE_PACKAGING)
    list(APPEND ERS_SUPERBUILD_CACHE_ARGS "-DERS_SUPERBUILD_ENABLE_PACKAGING:BOOL=ON")
endif()

ExternalProject_Add(BrainGenixERSProject
    SOURCE_DIR "${ERS_SUPERBUILD_SOURCE_DIR}"
    BINARY_DIR "${ERS_SUPERBUILD_BINARY_DIR}"
    DOWNLOAD_COMMAND ""
    UPDATE_COMMAND ""
    PATCH_COMMAND ""
    INSTALL_COMMAND ""
    TEST_COMMAND ""
    UPDATE_DISCONNECTED 1
    STEP_TARGETS configure build
    CMAKE_CACHE_ARGS ${ERS_SUPERBUILD_CACHE_ARGS}
)
